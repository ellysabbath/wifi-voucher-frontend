import { useState, useRef, useEffect } from 'react';
import { Link, useNavigate } from 'react-router-dom';
import {
  User, Mail, Lock, Eye, EyeOff, Phone, UserPlus,
  Camera, Upload, X, RefreshCw, CheckCircle2
} from 'lucide-react';
import AuthLayout from './AuthLayout';

export default function Register() {
  const navigate = useNavigate();
  const [showPassword, setShowPassword] = useState(false);
  const [photo, setPhoto] = useState<string>('');
  const [photoMode, setPhotoMode] = useState<'upload' | 'webcam' | null>(null);
  const [isCameraOn, setIsCameraOn] = useState(false);
  const [cameraError, setCameraError] = useState('');

  const videoRef = useRef<HTMLVideoElement>(null);
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const fileInputRef = useRef<HTMLInputElement>(null);
  const streamRef = useRef<MediaStream | null>(null);

  const [form, setForm] = useState({
    fullName: '',
    email: '',
    phone: '',
    password: '',
    confirmPassword: '',
  });
  const [error, setError] = useState('');

  // ===== WEBCAM =====
  const startCamera = async () => {
    setCameraError('');
    setPhotoMode('webcam');
    try {
      const stream = await navigator.mediaDevices.getUserMedia({
        video: { width: 640, height: 480, facingMode: 'user' },
      });
      streamRef.current = stream;
      if (videoRef.current) {
        videoRef.current.srcObject = stream;
        videoRef.current.play();
      }
      setIsCameraOn(true);
    } catch (err) {
      setCameraError('Imeshindwa kufungua kamera. Hakikisha umeruhusu kutumia kamera.');
      setIsCameraOn(false);
    }
  };

  const stopCamera = () => {
    if (streamRef.current) {
      streamRef.current.getTracks().forEach(track => track.stop());
      streamRef.current = null;
    }
    setIsCameraOn(false);
  };

  const capturePhoto = () => {
    if (videoRef.current && canvasRef.current) {
      const video = videoRef.current;
      const canvas = canvasRef.current;
      canvas.width = video.videoWidth;
      canvas.height = video.videoHeight;
      const ctx = canvas.getContext('2d');
      if (ctx) {
        ctx.drawImage(video, 0, 0);
        const dataUrl = canvas.toDataURL('image/jpeg', 0.9);
        setPhoto(dataUrl);
        stopCamera();
        setPhotoMode(null);
      }
    }
  };

  const retakePhoto = () => {
    setPhoto('');
    startCamera();
  };

  // ===== UPLOAD =====
  const handleFileUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;

    if (!file.type.startsWith('image/')) {
      setError('Tafadhali chagua faili la picha');
      return;
    }

    if (file.size > 5 * 1024 * 1024) {
      setError('Picha isizidi 5MB');
      return;
    }

    const reader = new FileReader();
    reader.onloadend = () => {
      setPhoto(reader.result as string);
      setPhotoMode(null);
      setError('');
    };
    reader.readAsDataURL(file);
  };

  // ===== CLEANUP =====
  useEffect(() => {
    return () => {
      stopCamera();
    };
  }, []);

  // ===== SUBMIT =====
  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    setError('');

    if (!form.fullName || !form.email || !form.phone || !form.password) {
      setError('Jaza sehemu zote');
      return;
    }
    if (form.password.length < 6) {
      setError('Nenosiri liwe na herufi 6 au zaidi');
      return;
    }
    if (form.password !== form.confirmPassword) {
      setError('Nenosiri hazifanani');
      return;
    }

    // Hapa unaweza kutuma `photo` (base64) kwenye API yako
    console.log('Photo:', photo);

    navigate('/verify-account', { state: { email: form.email } });
  };

  return (
    <AuthLayout title="Jisajili" subtitle="Unda akaunti yako ya mfumo wa kanisa">
      <form onSubmit={handleSubmit} className="space-y-4">
        {/* ===== PHOTO SECTION ===== */}
        <div className="flex flex-col items-center">
          <div className="relative">
            {photo ? (
              <div className="relative">
                <img
                  src={photo}
                  alt="Profile"
                  className="w-28 h-28 rounded-full object-cover border-4 border-primary shadow-lg"
                />
                <button
                  type="button"
                  onClick={() => setPhoto('')}
                  className="absolute -top-1 -right-1 bg-red-500 text-white p-1.5 rounded-full shadow-lg hover:bg-red-600 transition"
                  aria-label="Ondoa picha"
                >
                  <X size={14} />
                </button>
                <div className="absolute -bottom-1 -right-1 bg-green-500 text-white p-1.5 rounded-full shadow-lg">
                  <CheckCircle2 size={14} />
                </div>
              </div>
            ) : (
              <div className="w-28 h-28 rounded-full bg-gray-100 border-4 border-dashed border-gray-300 flex items-center justify-center text-gray-400">
                <User size={40} />
              </div>
            )}
          </div>

          {/* Buttons za kupakia/kuchukua picha */}
          {!photo && (
            <div className="flex gap-2 mt-3">
              <button
                type="button"
                onClick={() => fileInputRef.current?.click()}
                className="flex items-center gap-1.5 text-xs font-semibold text-primary bg-primary/10 hover:bg-primary/20 px-3 py-2 rounded-lg transition"
              >
                <Upload size={14} /> Pakia Picha
              </button>
              <button
                type="button"
                onClick={startCamera}
                className="flex items-center gap-1.5 text-xs font-semibold text-primary bg-primary/10 hover:bg-primary/20 px-3 py-2 rounded-lg transition"
              >
                <Camera size={14} /> Chukua Picha
              </button>
            </div>
          )}

          <input
            ref={fileInputRef}
            type="file"
            accept="image/*"
            onChange={handleFileUpload}
            className="hidden"
          />
        </div>

        {/* ===== WEBCAM MODAL ===== */}
        {photoMode === 'webcam' && (
          <div className="fixed inset-0 bg-black/70 z-50 flex items-center justify-center p-4">
            <div className="bg-white rounded-2xl shadow-2xl w-full max-w-lg overflow-hidden">
              <div className="flex items-center justify-between p-4 border-b">
                <h3 className="font-bold text-gray-800">Chukua Picha</h3>
                <button
                  type="button"
                  onClick={() => { stopCamera(); setPhotoMode(null); }}
                  className="p-2 hover:bg-gray-100 rounded-lg transition"
                >
                  <X size={20} />
                </button>
              </div>

              <div className="p-4">
                {cameraError ? (
                  <div className="bg-red-50 border border-red-200 text-red-700 text-sm rounded-lg p-4 text-center">
                    {cameraError}
                  </div>
                ) : (
                  <div className="relative bg-black rounded-xl overflow-hidden aspect-video">
                    <video
                      ref={videoRef}
                      autoPlay
                      playsInline
                      muted
                      className="w-full h-full object-cover"
                    />
                    {/* Guide circle */}
                    <div className="absolute inset-0 flex items-center justify-center pointer-events-none">
                      <div className="w-48 h-48 rounded-full border-4 border-white/50 border-dashed"></div>
                    </div>
                  </div>
                )}

                <canvas ref={canvasRef} className="hidden" />

                <div className="flex gap-3 justify-center mt-4">
                  {cameraError ? (
                    <button
                      type="button"
                      onClick={startCamera}
                      className="bg-primary text-white px-6 py-2.5 rounded-lg font-semibold hover:bg-primary-dark transition flex items-center gap-2"
                    >
                      <RefreshCw size={16} /> Jaribu Tena
                    </button>
                  ) : (
                    <button
                      type="button"
                      onClick={capturePhoto}
                      disabled={!isCameraOn}
                      className="bg-primary text-white px-6 py-2.5 rounded-lg font-semibold hover:bg-primary-dark transition flex items-center gap-2 disabled:opacity-50"
                    >
                      <Camera size={16} /> Chukua Picha
                    </button>
                  )}
                </div>
              </div>
            </div>
          </div>
        )}

        {/* ===== FORM FIELDS ===== */}
        <div>
          <label className="block text-sm font-semibold mb-1">Jina Kamili</label>
          <div className="relative">
            <User size={18} className="absolute left-3 top-3.5 text-gray-400" />
            <input
              value={form.fullName}
              onChange={e => setForm({ ...form, fullName: e.target.value })}
              placeholder="Jina lako kamili"
              className="w-full border rounded-lg pl-10 pr-3 py-2.5 focus:outline-none focus:ring-2 focus:ring-primary"
            />
          </div>
        </div>

        <div>
          <label className="block text-sm font-semibold mb-1">Barua Pepe</label>
          <div className="relative">
            <Mail size={18} className="absolute left-3 top-3.5 text-gray-400" />
            <input
              type="email"
              value={form.email}
              onChange={e => setForm({ ...form, email: e.target.value })}
              placeholder="jina@bahisda.org"
              className="w-full border rounded-lg pl-10 pr-3 py-2.5 focus:outline-none focus:ring-2 focus:ring-primary"
            />
          </div>
        </div>

        <div>
          <label className="block text-sm font-semibold mb-1">Namba ya Simu</label>
          <div className="relative">
            <Phone size={18} className="absolute left-3 top-3.5 text-gray-400" />
            <input
              value={form.phone}
              onChange={e => setForm({ ...form, phone: e.target.value })}
              placeholder="+255712345678"
              className="w-full border rounded-lg pl-10 pr-3 py-2.5 focus:outline-none focus:ring-2 focus:ring-primary"
            />
          </div>
        </div>

        <div>
          <label className="block text-sm font-semibold mb-1">Nenosiri</label>
          <div className="relative">
            <Lock size={18} className="absolute left-3 top-3.5 text-gray-400" />
            <input
              type={showPassword ? 'text' : 'password'}
              value={form.password}
              onChange={e => setForm({ ...form, password: e.target.value })}
              placeholder="••••••••"
              className="w-full border rounded-lg pl-10 pr-10 py-2.5 focus:outline-none focus:ring-2 focus:ring-primary"
            />
            <button
              type="button"
              onClick={() => setShowPassword(!showPassword)}
              className="absolute right-3 top-3 text-gray-400 hover:text-gray-600"
            >
              {showPassword ? <EyeOff size={18} /> : <Eye size={18} />}
            </button>
          </div>
        </div>

        <div>
          <label className="block text-sm font-semibold mb-1">Rudia Nenosiri</label>
          <div className="relative">
            <Lock size={18} className="absolute left-3 top-3.5 text-gray-400" />
            <input
              type={showPassword ? 'text' : 'password'}
              value={form.confirmPassword}
              onChange={e => setForm({ ...form, confirmPassword: e.target.value })}
              placeholder="••••••••"
              className="w-full border rounded-lg pl-10 pr-3 py-2.5 focus:outline-none focus:ring-2 focus:ring-primary"
            />
          </div>
        </div>

        {error && (
          <div className="bg-red-50 border border-red-200 text-red-700 text-sm rounded-lg p-3">
            {error}
          </div>
        )}

        <button
          type="submit"
          className="w-full bg-primary text-white py-3 rounded-lg font-semibold hover:bg-primary-dark transition flex items-center justify-center gap-2"
        >
          <UserPlus size={18} /> Jisajili
        </button>

        <p className="text-center text-sm text-gray-600">
          Una akaunti tayari?{' '}
          <Link to="/login" className="text-primary font-semibold hover:underline">
            Ingia
          </Link>
        </p>
      </form>
    </AuthLayout>
  );
}




import { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import { Mail, CheckCircle2 } from 'lucide-react';
import AuthLayout from './AuthLayout';

export default function VerifyAccount() {
  const navigate = useNavigate();
  const [code, setCode] = useState(['', '', '', '', '', '']);
  const [error, setError] = useState('');

  const handleChange = (index: number, value: string) => {
    if (!/^\d*$/.test(value)) return;
    const newCode = [...code];
    newCode[index] = value.slice(-1);
    setCode(newCode);

    // Auto-focus next input
    if (value && index < 5) {
      const next = document.getElementById(`code-${index + 1}`);
      next?.focus();
    }
  };

  const handleKeyDown = (index: number, e: React.KeyboardEvent) => {
    if (e.key === 'Backspace' && !code[index] && index > 0) {
      const prev = document.getElementById(`code-${index - 1}`);
      prev?.focus();
    }
  };

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    setError('');
    const fullCode = code.join('');
    if (fullCode.length !== 6) {
      setError('Ingiza msimbo kamili wa tarakimu 6');
      return;
    }
    // Mock verification - badilisha na API halisi
    navigate('/login');
  };

  const handleResend = () => {
    alert('Msimbo mpya umetumwa kwenye barua pepe yako');
  };

  return (
    <AuthLayout
      title="Thibitisha Akaunti"
      subtitle="Tumekutumia msimbo wa tarakimu 6 kwenye barua pepe yako"
    >
      <div className="flex justify-center mb-6">
        <div className="w-20 h-20 rounded-full bg-primary/10 flex items-center justify-center text-primary">
          <Mail size={36} />
        </div>
      </div>

      <form onSubmit={handleSubmit} className="space-y-6">
        <div>
          <label className="block text-sm font-semibold mb-3 text-center">
            Ingiza msimbo
          </label>
          <div className="flex justify-center gap-2">
            {code.map((digit, i) => (
              <input
                key={i}
                id={`code-${i}`}
                type="text"
                inputMode="numeric"
                value={digit}
                onChange={e => handleChange(i, e.target.value)}
                onKeyDown={e => handleKeyDown(i, e)}
                maxLength={1}
                className="w-12 h-14 text-center text-2xl font-bold border-2 rounded-lg focus:outline-none focus:border-primary focus:ring-2 focus:ring-primary/20"
              />
            ))}
          </div>
        </div>

        {error && (
          <div className="bg-red-50 border border-red-200 text-red-700 text-sm rounded-lg p-3 text-center">
            {error}
          </div>
        )}

        <button
          type="submit"
          className="w-full bg-primary text-white py-3 rounded-lg font-semibold hover:bg-primary-dark transition flex items-center justify-center gap-2"
        >
          <CheckCircle2 size={18} /> Thibitisha
        </button>

        <div className="text-center text-sm text-gray-600">
          Hukupata msimbo?{' '}
          <button
            type="button"
            onClick={handleResend}
            className="text-primary font-semibold hover:underline"
          >
            Tuma tena
          </button>
        </div>
      </form>
    </AuthLayout>
  );
}

import { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import { Lock, Eye, EyeOff, KeyRound } from 'lucide-react';
import AuthLayout from './AuthLayout';

export default function ResetPassword() {
  const navigate = useNavigate();
  const [showPassword, setShowPassword] = useState(false);
  const [form, setForm] = useState({ password: '', confirmPassword: '' });
  const [error, setError] = useState('');

  // Password strength
  const getStrength = (pwd: string) => {
    let score = 0;
    if (pwd.length >= 6) score++;
    if (pwd.length >= 10) score++;
    if (/[A-Z]/.test(pwd)) score++;
    if (/[0-9]/.test(pwd)) score++;
    if (/[^A-Za-z0-9]/.test(pwd)) score++;
    return score;
  };

  const strength = getStrength(form.password);
  const strengthLabels = ['Dhaifu sana', 'Dhaifu', 'Wastani', 'Nzuri', 'Nzuri sana', 'Imara'];
  const strengthColors = ['bg-red-500', 'bg-red-400', 'bg-yellow-500', 'bg-blue-500', 'bg-green-500', 'bg-green-600'];

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    setError('');

    if (form.password.length < 6) {
      setError('Nenosiri liwe na herufi 6 au zaidi');
      return;
    }
    if (form.password !== form.confirmPassword) {
      setError('Nenosiri hazifanani');
      return;
    }

    navigate('/password-reset-success');
  };

  return (
    <AuthLayout
      title="Weka Nenosiri Jipya"
      subtitle="Chagua nenosiri imara kwa akaunti yako"
    >
      <div className="flex justify-center mb-6">
        <div className="w-20 h-20 rounded-full bg-primary/10 flex items-center justify-center text-primary">
          <KeyRound size={36} />
        </div>
      </div>

      <form onSubmit={handleSubmit} className="space-y-4">
        <div>
          <label className="block text-sm font-semibold mb-1">Nenosiri Jipya</label>
          <div className="relative">
            <Lock size={18} className="absolute left-3 top-3.5 text-gray-400" />
            <input
              type={showPassword ? 'text' : 'password'}
              value={form.password}
              onChange={e => setForm({ ...form, password: e.target.value })}
              placeholder="••••••••"
              className="w-full border rounded-lg pl-10 pr-10 py-2.5 focus:outline-none focus:ring-2 focus:ring-primary"
            />
            <button
              type="button"
              onClick={() => setShowPassword(!showPassword)}
              className="absolute right-3 top-3 text-gray-400 hover:text-gray-600"
            >
              {showPassword ? <EyeOff size={18} /> : <Eye size={18} />}
            </button>
          </div>

          {/* Strength indicator */}
          {form.password && (
            <div className="mt-2">
              <div className="flex gap-1 mb-1">
                {[0, 1, 2, 3, 4].map(i => (
                  <div
                    key={i}
                    className={`h-1 flex-1 rounded-full ${
                      i < strength ? strengthColors[strength] : 'bg-gray-200'
                    }`}
                  ></div>
                ))}
              </div>
              <p className="text-xs text-gray-500">
                Nguvu: <span className="font-semibold">{strengthLabels[strength]}</span>
              </p>
            </div>
          )}
        </div>

        <div>
          <label className="block text-sm font-semibold mb-1">Rudia Nenosiri</label>

import { useState } from 'react';
import { useMemberStore } from '../store/memberStore';
import { useSMSStore } from '../store/smsStore';
import { Send, Users, User, Plus, Trash2 } from 'lucide-react';
import Modal from '../components/Modal';

export default function BulkSMS() {
  const members = useMemberStore(s => s.members);
  const sendSMS = useSMSStore(s => s.sendSMS);
  const messages = useSMSStore(s => s.messages);
  const groups = useSMSStore(s => s.groups);
  const addGroup = useSMSStore(s => s.addGroup);
  const deleteGroup = useSMSStore(s => s.deleteGroup);

  const [message, setMessage] = useState('');
  const [mode, setMode] = useState<'all' | 'group' | 'select'>('all');
  const [group, setGroup] = useState('');
  const [selected, setSelected] = useState<string[]>([]);
  const [success, setSuccess] = useState('');

  const [isGroupModalOpen, setIsGroupModalOpen] = useState(false);
  const [newGroupName, setNewGroupName] = useState('');
  const [newGroupMembers, setNewGroupMembers] = useState<string[]>([]);

  const ministries = [...new Set(members.map(m => m.ministry))];

  const getRecipients = () => {
    if (mode === 'all') return members.map(m => m.id);
    if (mode === 'group') {
      if (groups[group]) return groups[group];
      return members.filter(m => m.ministry === group).map(m => m.id);
    }
    return selected;
  };

  const handleSend = (e: React.FormEvent) => {
    e.preventDefault();
    const recipients = getRecipients();
    if (recipients.length === 0) return alert('Chagua wapokeaji');
    sendSMS(recipients, message, mode === 'group' ? group : mode === 'all' ? 'Wote' : 'Wateule');
    setSuccess(`SMS imetumwa kwa wanachama ${recipients.length}!`);
    setMessage('');
    setSelected([]);
    setTimeout(() => setSuccess(''), 3000);
  };

  const handleCreateGroup = (e: React.FormEvent) => {
    e.preventDefault();
    if (!newGroupName.trim() || newGroupMembers.length === 0) return;
    addGroup(newGroupName, newGroupMembers);
    setNewGroupName('');
    setNewGroupMembers([]);
    setIsGroupModalOpen(false);
  };

  const handleCloseGroupModal = () => {
    setNewGroupName('');
    setNewGroupMembers([]);
    setIsGroupModalOpen(false);
  };

  return (
    <div>
      <div className="flex justify-between items-center mb-6">
        <h1 className="text-3xl font-bold">Tuma SMS</h1>
        <button onClick={() => setIsGroupModalOpen(true)}
          className="bg-primary text-white px-4 py-2 rounded-lg flex items-center gap-2">
          <Plus size={18} /> Unda Group
        </button>
      </div>

      <Modal isOpen={isGroupModalOpen} onClose={handleCloseGroupModal} title="Unda Group Mpya">
        <form onSubmit={handleCreateGroup} className="space-y-4">
          <div>
            <label className="block text-sm font-semibold mb-1">Jina la Group *</label>
            <input required value={newGroupName}
              onChange={e => setNewGroupName(e.target.value)}
              placeholder="Mfano: Viongozi, Kwaya..."
              className="w-full border rounded-lg px-3 py-2" />
          </div>

          <div>
            <label className="block text-sm font-semibold mb-1">
              Chagua Wanachama ({newGroupMembers.length} wamechaguliwa) *
            </label>
            <div className="max-h-60 overflow-y-auto border rounded-lg p-2">
              {members.map(m => (
                <label key={m.id} className="flex items-center gap-2 p-2 hover:bg-gray-50">
                  <input type="checkbox"
                    checked={newGroupMembers.includes(m.id)}
                    onChange={e => setNewGroupMembers(prev =>
                      e.target.checked ? [...prev, m.id] : prev.filter(i => i !== m.id)
                    )} />
                  <span>{m.fullName} - {m.phone}</span>
                </label>
              ))}
            </div>
          </div>

          <div className="flex gap-3 justify-end pt-4 border-t">
            <button type="button" onClick={handleCloseGroupModal}
              className="bg-gray-200 px-6 py-2 rounded-lg hover:bg-gray-300">
              Ghairi
            </button>
            <button type="submit"
              disabled={!newGroupName.trim() || newGroupMembers.length === 0}
              className="bg-primary text-white px-6 py-2 rounded-lg hover:bg-primary-dark disabled:opacity-50">
              Unda Group
            </button>
          </div>
        </form>
      </Modal>

      <form onSubmit={handleSend} className="bg-white rounded-lg shadow p-6 mb-6">
        <div className="mb-4">
          <label className="block text-sm font-semibold mb-2">Wapokeaji</label>
          <div className="flex gap-2 mb-3 flex-wrap">
            <button type="button" onClick={() => setMode('all')}
              className={`px-4 py-2 rounded-lg flex items-center gap-2 ${mode === 'all' ? 'bg-primary text-white' : 'bg-gray-200'}`}>
              <Users size={16} /> Wote
            </button>
            <button type="button" onClick={() => setMode('group')}
              className={`px-4 py-2 rounded-lg ${mode === 'group' ? 'bg-primary text-white' : 'bg-gray-200'}`}>
              Kwa Group
            </button>
            <button type="button" onClick={() => setMode('select')}
              className={`px-4 py-2 rounded-lg flex items-center gap-2 ${mode === 'select' ? 'bg-primary text-white' : 'bg-gray-200'}`}>
              <User size={16} /> Chagua
            </button>
          </div>

          {mode === 'group' && (
            <div className="space-y-2">
              <select value={group} onChange={e => setGroup(e.target.value)}
                className="w-full border rounded-lg px-3 py-2">
                <option value="">-- Chagua Group --</option>
                <optgroup label="Groups Zilizoundwa">
                  {Object.keys(groups).map(g => (
                    <option key={g} value={g}>{g} ({groups[g].length})</option>
                  ))}
                </optgroup>
                <optgroup label="Huduma">
                  {ministries.map(m => (
                    <option key={m} value={m}>{m}</option>
                  ))}
                </optgroup>
              </select>

              {groups[group] && (
                <button type="button" onClick={() => deleteGroup(group)}
                  className="text-red-500 text-sm flex items-center gap-1">
                  <Trash2 size={14} /> Futa group hii
                </button>
              )}
            </div>
          )}

          {mode === 'select' && (
            <div className="max-h-60 overflow-y-auto border rounded-lg p-2">
              {members.map(m => (
                <label key={m.id} className="flex items-center gap-2 p-2 hover:bg-gray-50">
                  <input type="checkbox" checked={selected.includes(m.id)}
                    onChange={e => setSelected(prev =>
                      e.target.checked ? [...prev, m.id] : prev.filter(i => i !== m.id)
                    )} />
                  <span>{m.fullName} - {m.phone}</span>
                </label>
              ))}
            </div>
          )}
        </div>

        <div className="mb-4">
          <label className="block text-sm font-semibold mb-2">Ujumbe</label>
          <textarea required value={message}
            onChange={e => setMessage(e.target.value)} rows={4}
            className="w-full border rounded-lg px-3 py-2"
            placeholder="Andika ujumbe wako hapa..." />
          <p className="text-sm text-gray-500 mt-1">{message.length} herufi</p>
        </div>

        <button type="submit"
          className="bg-primary text-white px-6 py-3 rounded-lg flex items-center gap-2 hover:bg-primary-dark">
          <Send size={18} /> Tuma SMS
        </button>

        {success && <p className="mt-4 text-green-600 font-semibold">{success}</p>}
      </form>

      <div className="bg-white rounded-lg shadow p-6">
        <h2 className="text-xl font-bold mb-4">Historia ya SMS</h2>
        <div className="space-y-3">
          {messages.map(m => (
            <div key={m.id} className="border-l-4 border-primary pl-3 py-2">
              <p className="text-sm text-gray-500">
                {new Date(m.sentAt).toLocaleString()} - Kwa: {m.group || `${m.recipients.length} wateule`}
              </p>
              <p>{m.message}</p>
              <span className="text-xs text-green-600">{m.status}</span>
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}

import { useState, useRef, useEffect } from 'react';
import { useChatStore } from '../store/chatStore';
import { Send, Plus, Trash2, Hash } from 'lucide-react';
import Modal from '../components/Modal';

export default function Chat() {
  const messages = useChatStore(s => s.messages);
  const rooms = useChatStore(s => s.rooms);
  const addRoom = useChatStore(s => s.addRoom);
  const deleteRoom = useChatStore(s => s.deleteRoom);
  const sendMessage = useChatStore(s => s.sendMessage);
  const [text, setText] = useState('');
  const [room, setRoom] = useState('general');
  const [isModalOpen, setIsModalOpen] = useState(false);
  const [roomName, setRoomName] = useState('');
  const endRef = useRef<HTMLDivElement>(null);

  const filtered = messages.filter(m => m.room === room);

  useEffect(() => {
    endRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [filtered]);

  const handleSend = (e: React.FormEvent) => {
    e.preventDefault();
    if (!text.trim()) return;
    sendMessage('current-user', 'Mimi', text, room);
    setText('');
  };

  const handleCreateRoom = (e: React.FormEvent) => {
    e.preventDefault();
    if (!roomName.trim()) return;
    addRoom(roomName);
    setRoom(roomName.toLowerCase().replace(/\s+/g, '-'));
    setRoomName('');
    setIsModalOpen(false);
  };

  const handleCloseModal = () => {
    setRoomName('');
    setIsModalOpen(false);
  };

  return (
    <div>
      <div className="flex justify-between items-center mb-6">
        <h1 className="text-3xl font-bold">Mazungumzo</h1>
        <button onClick={() => setIsModalOpen(true)}
          className="bg-primary text-white px-4 py-2 rounded-lg flex items-center gap-2">
          <Plus size={18} /> Unda Room Mpya
        </button>
      </div>

      <Modal isOpen={isModalOpen} onClose={handleCloseModal} title="Unda Room Mpya">
        <form onSubmit={handleCreateRoom} className="space-y-4">
          <div>
            <label className="block text-sm font-semibold mb-1">Jina la Room *</label>
            <input required value={roomName}
              onChange={e => setRoomName(e.target.value)}
              placeholder="Mfano: Kwaya, Uinjilisti..."
              className="w-full border rounded-lg px-3 py-2" />
            <p className="text-xs text-gray-500 mt-1">
              Room itaonekana kama: #{roomName.toLowerCase().replace(/\s+/g, '-') || 'jina-la-room'}
            </p>
          </div>
          <div className="flex gap-3 justify-end pt-4 border-t">
            <button type="button" onClick={handleCloseModal}
              className="bg-gray-200 px-6 py-2 rounded-lg hover:bg-gray-300">
              Ghairi
            </button>
            <button type="submit"
              className="bg-primary text-white px-6 py-2 rounded-lg hover:bg-primary-dark">
              Unda
            </button>
          </div>
        </form>
      </Modal>

      <div className="flex gap-2 mb-4 flex-wrap">
        {rooms.map(r => (
          <div key={r} className="relative group">
            <button onClick={() => setRoom(r)}
              className={`px-4 py-2 rounded-lg text-sm flex items-center gap-1 ${room === r ? 'bg-primary text-white' : 'bg-white'}`}>
              <Hash size={14} /> {r}
            </button>
            {r !== 'general' && (
              <button onClick={() => deleteRoom(r)}
                className="absolute -top-2 -right-2 bg-red-500 text-white rounded-full p-1 opacity-0 group-hover:opacity-100 transition">
                <Trash2 size={10} />
              </button>
            )}
          </div>
        ))}
      </div>

      <div className="bg-white rounded-lg shadow flex flex-col h-[600px]">
        <div className="flex-1 overflow-y-auto p-4 space-y-3">
          {filtered.length === 0 ? (
            <p className="text-center text-gray-400 mt-8">Hakuna ujumbe bado. Anza mazungumzo!</p>
          ) : (
            filtered.map(m => (
              <div key={m.id} className={`flex ${m.senderId === 'current-user' ? 'justify-end' : 'justify-start'}`}>
                <div className={`max-w-[70%] rounded-lg p-3 ${m.senderId === 'current-user' ? 'bg-primary text-white' : 'bg-gray-100'}`}>
                  <p className="text-xs font-semibold mb-1">{m.senderName}</p>
                  <p>{m.message}</p>
                  <p className="text-xs opacity-70 mt-1">
                    {new Date(m.timestamp).toLocaleTimeString()}
                  </p>
                </div>
              </div>
            ))
          )}
          <div ref={endRef} />
        </div>

        <form onSubmit={handleSend} className="border-t p-4 flex gap-2">
          <input
            value={text}
            onChange={e => setText(e.target.value)}
            placeholder={`Andika ujumbe kwenye #${room}...`}
            className="flex-1 border rounded-lg px-3 py-2"
          />
          <button type="submit" className="bg-primary text-white px-4 rounded-lg">
            <Send size={18} />
          </button>
        </form>
      </div>
    </div>
  );
}

import { X } from 'lucide-react';
import { useEffect } from 'react';

interface ModalProps {
  isOpen: boolean;
  onClose: () => void;
  title: string;
  children: React.ReactNode;
}

export default function Modal({ isOpen, onClose, title, children }: ModalProps) {
  // Close on Escape key
  useEffect(() => {
    const handleEsc = (e: KeyboardEvent) => {
      if (e.key === 'Escape') onClose();
    };
    if (isOpen) document.addEventListener('keydown', handleEsc);
    return () => document.removeEventListener('keydown', handleEsc);
  }, [isOpen, onClose]);

  // Prevent body scroll when open
  useEffect(() => {
    if (isOpen) document.body.style.overflow = 'hidden';
    else document.body.style.overflow = 'unset';
    return () => { document.body.style.overflow = 'unset'; };
  }, [isOpen]);

  if (!isOpen) return null;

  return (
    <div
      className="fixed inset-0 bg-black/50 z-50 flex items-center justify-center p-4"
      onClick={onClose}
    >
      <div
        className="bg-white rounded-lg shadow-xl w-full max-w-2xl max-h-[90vh] overflow-y-auto"
        onClick={(e) => e.stopPropagation()}
      >
        {/* Header */}
        <div className="flex justify-between items-center p-4 border-b sticky top-0 bg-white z-10">
          <h2 className="text-xl font-bold">{title}</h2>
          <button
            onClick={onClose}
            className="p-2 hover:bg-gray-100 rounded-lg transition"
            aria-label="Funga"
          >
            <X size={20} />
          </button>
        </div>

        {/* Body */}
        <div className="p-6">{children}</div>
      </div>
    </div>
  );
}

interface ChatState {
  messages: ChatMessage[];
  rooms: string[];
  addRoom: (name: string) => void;
  deleteRoom: (name: string) => void;
  sendMessage: (senderId: string, senderName: string, message: string, room: string) => void;
}

export const useChatStore = create<ChatState>((set, get) => ({
  messages: initialMessages,
  rooms: ['general', 'vijana', 'wanawake', 'wazee', 'uongozi'],

  addRoom: (name) => {
    const clean = name.toLowerCase().replace(/\s+/g, '-');
    if (get().rooms.includes(clean)) return;
    set({ rooms: [...get().rooms, clean] });
  },

  deleteRoom: (name) => set({
    rooms: get().rooms.filter(r => r !== name),
    messages: get().messages.filter(m => m.room !== name),
  }),

  sendMessage: (senderId, senderName, message, room) => set({
    messages: [...get().messages, {
      id: crypto.randomUUID(),
      senderId,
      senderName,
      message,
      room,
      timestamp: new Date().toISOString(),
    }]
  }),
}));
